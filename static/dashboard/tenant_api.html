Code examples
Push car stock to ChatPlatform
// Next.js API route — push stock when cars update
const res = await fetch('https://web-production-f61d6.up.railway.app/api/sync/stock', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'X-Sync-Key': 'YOUR_SYNC_KEY',
  },
  body: JSON.stringify({
    cars: [
      {
        car_id: 'PROTON-SAGA-2024',
        brand: 'Proton', model: 'Saga', variant: 'Premium CVT',
        year: 2024, price_otr: 48800, stock: 7,
        status: 'available', transmission: 'CVT',
      }
    ]
  })
});
Receive leads from ChatPlatform (webhook handler)
// Next.js API route — app/api/chatbot/leads/route.ts
export async function POST(req: Request) {
  const syncKey = req.headers.get('x-sync-key');
  if (syncKey !== process.env.CHATPLATFORM_SYNC_KEY) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const lead = await req.json();
  // lead = { name, phone, interest, session_id, created_at }

  await saveLeadToFirebase(lead); // your existing function
  return Response.json({ ok: true });
}
Poll for leads manually
// GET new unnotified leads
const res = await fetch('https://web-production-f61d6.up.railway.app/api/sync/leads', {
  headers: { 'X-Sync-Key': 'YOUR_SYNC_KEY' }
});
const { leads } = await res.json();

// Mark each lead as received
for (const lead of leads) {
  await fetch(
    `https://web-production-f61d6.up.railway.app/api/sync/leads/${lead.id}/notified`,
    { method: 'PATCH', headers: { 'X-Sync-Key': 'YOUR_SYNC_KEY' } }
  );
}
